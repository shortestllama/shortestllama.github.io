```python
from itertools import product, islice
import json
from pwn import remote

HOST = "chal.sunshinectf.games"
PORT = 25201

L = 10          # genes per sample
POP = 100       # samples per population (service requires 100)
GEN_LIMIT = 100 # number of submissions to send; set to None to send all 1024

def populations_binary_rows():
    """Yield populations where each is 100 copies of a single binary row (as floats)."""
    for comb in product([0, 1], repeat=L):
        row = [float(x) for x in comb]
        yield [row] * POP, row

def drain_lines(r, timeout=1.0):
    """Read and print whatever the server sends for a bit."""
    printed_any = False
    while True:
        try:
            line = r.recvline(timeout=timeout)
            if not line:
                break
            s = line.decode(errors="replace").rstrip()
            print("->", s)
            printed_any = True
            # Heuristic: stop early if we see a generation result line or terminal messages
            if any(k in s.lower() for k in ["expected 100 samples", "evolution failed", "success", "average", "survive"]):
                # keep draining a bit more to catch multi-line results
                continue
        except EOFError:
            print("[!] server closed connection (EOF)")
            return False
        except Exception:
            break
    return True

def main():
    r = remote(HOST, PORT)
    # Read banner / intro
    drain_lines(r, timeout=2.0)

    gen_iter = populations_binary_rows()
    if GEN_LIMIT is not None:
        gen_iter = islice(gen_iter, GEN_LIMIT)

    for idx, (pop, row) in enumerate(gen_iter, 1):
        payload = json.dumps({"samples": pop}).encode("ascii")
        print(f"[{idx:03d}] sending row -> {row}")
        try:
            r.sendline(payload)
        except EOFError:
            print("[!] server closed connection on send; stopping")
            break

        if not drain_lines(r, timeout=1.0):
            break

    try:
        r.interactive()  # if server stays open, drop you into interactive
    except Exception:
        pass
    try:
        r.close()
    except Exception:
        pass

if __name__ == "__main__":
    main()
```
